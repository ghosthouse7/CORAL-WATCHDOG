Incident Analysis — Docker + GitHub JOIN
SELECT dc."Image", dc."Status", gp.title AS last_pr, gp.user__login AS author
FROM docker.containers dc
LEFT JOIN github.pulls gp
  ON gp.owner = 'ghosthouse7' AND gp.repo = 'CORAL-WATCHDOG'
WHERE dc."State" != 'running'
LIMIT 10
